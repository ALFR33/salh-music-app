const TOTAL_TRACKS = 30000000000;
const PAGE_SIZE = 40;
const YOUTUBE_URL = "https://www.youtube.com/@SalhAlheibKlel";
const FACEBOOK_URL = "https://www.facebook.com/salhalheibkle";

export default function Component() {
  const [activeCategory, setActiveCategory] = useState("all");
  const [searchQuery, setSearchQuery] = useState("");
  const [currentPage, setCurrentPage] = useState(1);
  const [currentTrack, setCurrentTrack] = useState<Track | null>(ORIGINAL_TRACKS[0]);
  const [isPlaying, setIsPlaying] = useState(false);
  const [volume, setVolume] = useState(0.8);
  const [progress, setProgress] = useState(0);
  const [duration, setDuration] = useState(0);
  const [copiedLink, setCopiedLink] = useState<string | null>(null);

  const audioRef = useRef<HTMLAudioElement | null>(null);

  const getAudioUrl = (clipId: string) => {
    return `https://storage.googleapis.com/producer-app-public/clips/${clipId}.m4a`;
  };

  const currentCategoryKeys = useMemo(() => {
    if (activeCategory === "all") {
      return Object.keys(GENRE_PREFIXES);
    }
    return [activeCategory];
  }, [activeCategory]);

  const pageTracks = useMemo(() => {
    const tracks: Track[] = [];

    if (currentPage === 1 && searchQuery === "") {
      ORIGINAL_TRACKS.forEach(t => {
        if (activeCategory === "all" || t.category === activeCategory) {
          tracks.push(t);
        }
      });
    }

    const startIndex = (currentPage - 1) * PAGE_SIZE;
    const catKeys = currentCategoryKeys;

    for (let i = 0; i < PAGE_SIZE; i++) {
      const globalIdx = startIndex + i + 1;
      const key = catKeys[globalIdx % catKeys.length];
      const list = GENRE_PREFIXES[key] || GENRE_PREFIXES.cinema;
      const prefix = list[globalIdx % list.length];

      let categoryLabel = "عالمي وسينمائي";
      if (key === "native") categoryLabel = "الهنود الحمر والقبايل";
      if (key === "cinema") categoryLabel = "أفلام ومسلسلات عالمية";
      if (key === "arab_tv") categoryLabel = "دراما عربية ومصرية وسورية وسعودية";
      if (key === "world") categoryLabel = "موسيقى الشعوب والدول";
      if (key === "relax") categoryLabel = "استرخاء وجمال الكون";
      if (key === "trance") categoryLabel = "ترانس وريمكس ودي جي";
      if (key === "war") categoryLabel = "قتالية وحربيات وعسكرية";
      if (key === "hiphop") categoryLabel = "Hip Hop & Beats";
      if (key === "oriental") categoryLabel = "يرغول وشرقي أصيل";
      if (key === "cinematic") categoryLabel = "تيتانيك وملحمي";

      const clipId = TEMPLATE_CLIPS[globalIdx % TEMPLATE_CLIPS.length];
      const title = `${prefix} #${globalIdx.toLocaleString()}`;

      if (searchQuery === "" || title.toLowerCase().includes(searchQuery.toLowerCase()) || categoryLabel.toLowerCase().includes(searchQuery.toLowerCase())) {
        tracks.push({
          id: `proc-${globalIdx}`,
          title,
          category: key,
          categoryLabel,
          clipId
        });
      }
    }

    return tracks;
  }, [currentPage, activeCategory, searchQuery, currentCategoryKeys]);

  const totalPages = useMemo(() => {
    return Math.floor(TOTAL_TRACKS / PAGE_SIZE);
  }, []);

  useEffect(() => {
    if (audioRef.current) {
      audioRef.current.volume = volume;
    }
  }, [volume]);

  const handlePlayTrack = (track: Track) => {
    if (currentTrack?.id === track.id) {
      if (isPlaying) {
        audioRef.current?.pause();
        setIsPlaying(false);
      } else {
        audioRef.current?.play();
        setIsPlaying(true);
      }
    } else {
      setCurrentTrack(track);
      setIsPlaying(true);
      if (audioRef.current) {
        audioRef.current.src = getAudioUrl(track.clipId);
        audioRef.current.play().catch(() => setIsPlaying(false));
      }
    }
  };

  const copyToClipboard = (text: string, type: string) => {
    navigator.clipboard.writeText(text);
    setCopiedLink(type);
    setTimeout(() => setCopiedLink(null), 2500);
  };

  const handleTimeUpdate = () => {
    if (audioRef.current) {
      setProgress(audioRef.current.currentTime);
      setDuration(audioRef.current.duration || 0);
    }
  };

  const formatTime = (time: number) => {
    if (isNaN(time)) return "0:00";
    const mins = Math.floor(time / 60);
    const secs = Math.floor(time % 60);
    return `${mins}:${secs < 10 ? "0" : ""}${secs}`;
  };

  return (
    <div className="min-h-screen w-full flex flex-col items-center bg-slate-950 text-slate-100 font-sans pb-36 dir-rtl">
      <header className="w-full bg-gradient-to-r from-slate-900 via-blue-950 to-slate-900 border-b border-slate-800 py-10 px-4 text-center relative overflow-hidden">

        <div className="max-w-5xl mx-auto flex flex-col items-center gap-4 relative z-10">
          <div className="flex items-center gap-2 px-4 py-1.5 rounded-full bg-blue-500/10 border border-blue-500/30 text-blue-400 text-xs font-semibold uppercase tracking-wider">
            <ShieldCheck className="w-4 h-4" /> المكتبة الفنية الرسمية الشاملة
          </div>
          <h1 className="text-4xl md:text-5xl font-black bg-gradient-to-r from-blue-400 via-indigo-300 to-sky-400 bg-clip-text text-transparent">
            Salh Alheib Klel
          </h1>
          <p className="text-slate-400 text-sm md:text-base max-w-2xl">
            أرشيف الموسيقى العالمي والشامل (30 مليار مقطوعة) • استرخاء، ترانس، هيب هوب، دراما عربية وعالمية، قتالية، موسيقى الشعوب، واستماع وتحميل مباشر
          </p>
          <div className="flex flex-wrap items-center justify-center gap-3 mt-2">
            <button onClick={() => window.open(YOUTUBE_URL, "_blank", "noopener,noreferrer")} className="flex items-center gap-2 px-5 py-2.5 rounded-full bg-red-600 hover:bg-red-500 text-white font-bold text-xs md:text-sm shadow-lg transition-all">
              <Youtube className="w-4 h-4 fill-white" /> القناة الرسمية على YouTube
              <ExternalLink className="w-3.5 h-3.5 opacity-70" />
            </button>
            <button onClick={() => window.open(FACEBOOK_URL, "_blank", "noopener,noreferrer")} className="flex items-center gap-2 px-5 py-2.5 rounded-full bg-blue-600 hover:bg-blue-500 text-white font-bold text-xs md:text-sm shadow-lg transition-all">
              <Facebook className="w-4 h-4 fill-white" /> الصفحة الرسمية على Facebook
              <ExternalLink className="w-3.5 h-3.5 opacity-70" />
            </button>
          </div>
        </div>
      </header>

      <main className="w-full max-w-7xl px-4 py-8 flex flex-col gap-6">
        <div className="flex flex-col md:flex-row items-center justify-between gap-4 bg-slate-900/90 border border-slate-800 p-4 rounded-2xl">
          <div className="relative w-full md:w-96">
            <Search className="absolute right-3.5 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-500" />
            <input
              type="text"
              placeholder="ابحث في الأرشيف..."
              value={searchQuery}
              onChange={(e) => { setSearchQuery(e.target.value); setCurrentPage(1); }}
              className="w-full bg-slate-950 border border-slate-800 text-slate-100 pr-10 pl-4 py-2.5 rounded-xl text-sm focus:outline-none focus:border-blue-500"
            />
          </div>
          <div className="flex items-center gap-3 text-xs text-slate-400 font-mono bg-slate-950/60 px-4 py-2.5 rounded-xl border border-slate-800">
            <Database className="w-4 h-4 text-blue-400" />
            <span>الإجمالي: <strong className="text-blue-400">30,000,000,000</strong> مقطوعة</span>
          </div>
        </div>

        <div className="flex items-center gap-2 overflow-x-auto pb-2 scrollbar-none">
          {CATEGORIES.map((cat) => {
            const Icon = cat.icon;
            const isActive = activeCategory === cat.id;
            return (
              <button
                key={cat.id}
                onClick={() => { setActiveCategory(cat.id); setCurrentPage(1); }}
                className={`flex items-center gap-2 px-4 py-2.5 rounded-xl text-xs font-semibold whitespace-nowrap transition-all border ${
                  isActive ? "bg-blue-600 text-white border-blue-500 shadow-md" : "bg-slate-900/80 text-slate-400 border-slate-800 hover:bg-slate-800 hover:text-slate-200"
                }`}
              >
                <Icon className="w-4 h-4" /> {cat.name}
              </button>
            );
          })}
        </div>

        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4 mt-2">
          {pageTracks.map((track) => {
            const isSelected = currentTrack?.id === track.id;
            const isTrackPlaying = isSelected && isPlaying;
            return (
              <div key={track.id} className={`flex flex-col justify-between p-5 rounded-2xl border transition-all ${isSelected ? "bg-gradient-to-b from-blue-950/40 to-slate-900 border-blue-500/50 shadow-lg" : "bg-slate-900/60 border-slate-800 hover:border-slate-700"}`}>
                <div className="flex items-start justify-between gap-3 mb-4">
                  <div>
                    <div className="flex items-center gap-2 mb-1.5">
                      <span className="text-[10px] font-bold px-2 py-0.5 rounded-full bg-blue-500/10 text-blue-400 border border-blue-500/20">{track.categoryLabel}</span>
                      {track.isOriginal && <span className="text-[10px] font-bold px-2 py-0.5 rounded-full bg-emerald-500/10 text-emerald-400 border border-emerald-500/20">أصلي 100%</span>}
                    </div>
                    <h3 className="text-base font-bold text-slate-100 line-clamp-1">{track.title}</h3>
                    <p className="text-xs text-slate-500 mt-1 flex items-center gap-1">
                      <ShieldCheck className="w-3.5 h-3.5 text-blue-400" /> © Salh Alheib Klel
                    </p>
                  </div>
                  <button onClick={() => handlePlayTrack(track)} className={`w-11 h-11 rounded-full flex items-center justify-center shrink-0 transition-transform active:scale-90 ${isTrackPlaying ? "bg-blue-600 text-white shadow-md" : "bg-slate-800 hover:bg-blue-600 text-slate-200 hover:text-white"}`}>
                    {isTrackPlaying ? <Pause className="w-5 h-5 fill-current" /> : <Play className="w-5 h-5 fill-current ml-0.5" />}
                  </button>
                </div>
                <div className="flex items-center gap-2 pt-3 border-t border-slate-800/60">
                  <a href={getAudioUrl(track.clipId)} download={`${track.title}.m4a`} className="flex-1 flex items-center justify-center gap-2 py-2 rounded-xl bg-slate-950 border border-slate-800 hover:border-slate-700 text-slate-300 hover:text-white text-xs font-medium transition-colors">
                    <Download className="w-3.5 h-3.5" /> تحميل M4A / MP3
                  </a>
                </div>
              </div>
            );
          })}
        </div>

        <div className="flex items-center justify-between gap-4 mt-6 bg-slate-900/90 border border-slate-800 p-4 rounded-2xl">
          <button disabled={currentPage <= 1} onClick={() => setCurrentPage(prev => Math.max(1, prev - 1))} className="flex items-center gap-2 px-4 py-2 rounded-xl bg-slate-950 border border-slate-800 text-slate-300 hover:text-white text-xs font-semibold disabled:opacity-40">
            <SkipBack className="w-4 h-4" /> الصفحة السابقة
          </button>
          <span className="text-xs text-slate-400 font-mono">صفحة <strong className="text-blue-400">{currentPage.toLocaleString()}</strong> من <strong className="text-slate-300">{totalPages.toLocaleString()}</strong></span>
          <button disabled={currentPage >= totalPages} onClick={() => setCurrentPage(prev => Math.min(totalPages, prev + 1))} className="flex items-center gap-2 px-4 py-2 rounded-xl bg-slate-950 border border-slate-800 text-slate-300 hover:text-white text-xs font-semibold disabled:opacity-40">
            الصفحة التالية <SkipForward className="w-4 h-4" />
          </button>
        </div>
      </main>

      {currentTrack && (
        <div className="fixed bottom-0 left-0 w-full bg-slate-900/95 border-t border-slate-800 p-4 backdrop-blur-xl z-50 flex flex-col md:flex-row items-center justify-between gap-4 shadow-2xl">
          <audio ref={audioRef} src={getAudioUrl(currentTrack.clipId)} onTimeUpdate={handleTimeUpdate} onEnded={() => setIsPlaying(false)} crossOrigin="anonymous" />
          <div className="flex items-center gap-3 w-full md:w-auto">
            <div className="w-10 h-10 rounded-xl bg-blue-600/20 border border-blue-500/30 flex items-center justify-center shrink-0">
              <Music className="w-5 h-5 text-blue-400 animate-pulse" />
            </div>
            <div>
              <h4 className="text-sm font-bold text-slate-100 line-clamp-1">{currentTrack.title}</h4>
              <p className="text-xs text-blue-400 flex items-center gap-1">
                <ShieldCheck className="w-3 h-3" /> Salh Alheib Klel • {currentTrack.categoryLabel}
              </p>
            </div>
          </div>
          <div className="flex flex-col items-center gap-2 w-full md:w-1/2 max-w-xl">
            <button onClick={() => handlePlayTrack(currentTrack)} className="w-10 h-10 rounded-full bg-blue-600 hover:bg-blue-500 text-white flex items-center justify-center shadow-lg">
              {isPlaying ? <Pause className="w-5 h-5 fill-current" /> : <Play className="w-5 h-5 fill-current ml-0.5" />}
            </button>
            <div className="w-full flex items-center gap-3 text-xs text-slate-400 font-mono">
              <span>{formatTime(progress)}</span>
              <input type="range" min="0" max={duration || 100} value={progress} onChange={(e) => { const val = Number(e.target.value); setProgress(val); if (audioRef.current) audioRef.current.currentTime = val; }} className="w-full accent-blue-500 bg-slate-950 rounded-lg h-1.5 cursor-pointer" />
              <span>{formatTime(duration)}</span>
            </div>
          </div>
        </div>
      )}
